เชื่อมต่อ Google Sheets (Web App)

ส่งข้อมูลรายรับ รายจ่าย การลงทุน และหนี้สินไปยังชีตของคุณ

โหมดการส่งข้อมูลขึ้น Sheets:

สถานะล่าสุด: ยังไม่เคยส่งหรือดึงข้อมูล

⚠️ ถ้ากดส่งแล้วไม่มีข้อมูลเข้าชีต หรือดึงข้อมูลไม่ได้ ให้เช็ก 4 ข้อนี้:

1. URL ต้องลงท้ายด้วย /exec (ไม่ใช่ /dev)

2. ใน Apps Script ตอน Deploy ต้องเลือก Who has access: Anyone และทุกครั้งที่แก้โค้ด Script ต้องสร้าง New deployment

3. สคริปต์จะสร้างแท็บชื่อ Minnie_App_Sync และ Minnie_App_State อัตโนมัติในไฟล์ Google Sheets ของคุณ

4. หากแก้ไขข้อมูลใน Google Sheets เอง แล้วกด ดึงจาก Sheets ข้อมูลในแอปและกราฟทั้งหมดจะอัปเดตตามทันที

วิธีติดตั้งสคริปต์ใน Google Sheets (2-Way Sync):
  1. เปิดไฟล์ Google Sheets ของคุณ แล้วไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. วางโค้ดด้านล่างลงไป แล้วกดบันทึกและ ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่
  3. เลือกประเภทเป็น เว็บแอป (Web app), สิทธิ์การเข้าถึงเป็น ทุกคน (Anyone) แล้วนำ URL มาวางด้านบน
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const stateSheet = ss.getSheetByName('Minnie_App_State');
    let state = null;
    if (stateSheet && stateSheet.getLastRow() >= 2) {
      const rawJson = stateSheet.getRange(2, 2).getValue();
      if (rawJson) state = JSON.parse(rawJson);
    }
    const payload = JSON.stringify({ status: 'ok', state: state });
    const callback = e && e.parameter && e.parameter.callback;
    if (callback) {
      return ContentService
        .createTextOutput(callback + '(' + payload + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService
      .createTextOutput(payload)
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    const errPayload = JSON.stringify({ status: 'error', message: err.message });
    const callback = e && e.parameter && e.parameter.callback;
    if (callback) {
      return ContentService
        .createTextOutput(callback + '(' + errPayload + ');')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService
      .createTextOutput(errPayload)
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const data = JSON.parse(raw);
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const now = new Date();
    const batchId = data.batchId || Utilities.getUuid().slice(0, 8);
    const mode = data.syncMode || 'replace_all';

    // 1. Save full state JSON so pulling back into the app restores exact records, prices, debts & budgets
    if (data.fullState) {
      let stateSheet = ss.getSheetByName('Minnie_App_State');
      if (!stateSheet) {
        stateSheet = ss.insertSheet('Minnie_App_State');
        stateSheet.appendRow(['Updated_At', 'State_JSON']);
      }
      stateSheet.getRange(2, 1, 1, 2).setValues([[now, JSON.stringify(data.fullState)]]);
    }

    // 2. Human-readable rows in Minnie_App_Sync
    let sheet = ss.getSheetByName('Minnie_App_Sync');
    if (!sheet) {
      sheet = ss.insertSheet('Minnie_App_Sync');
    }
    const headers = ['Synced_At', 'Batch_Id', 'Kind', 'Type/Action', 'Date', 'YearMonth', 'Name/Category', 'Group', 'Amount', 'QTY', 'Note'];
    const rows = [];

    if (data.kind === 'ping') {
      rows.push([now, batchId, 'ping', 'test', Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd'), '', 'Connection Test', '-', 0, 0, 'Ping from Minnie App']);
    }

    (data.transactions || []).forEach(t => {
      rows.push([now, batchId, 'transaction', t.type, t.date, t.yearMonth, t.category, '-', Number(t.amount || 0), 0, t.note || '']);
    });
    (data.investments || []).forEach(i => {
      rows.push([now, batchId, 'investment', i.action, i.date, i.yearMonth, i.asset, i.group, Number(i.amount || 0), Number(i.qty || 0), 'Portfolio']);
    });
    (data.debts || []).forEach(d => {
      rows.push([now, batchId, 'debt', 'snapshot', Utilities.formatDate(now, Session.getScriptTimeZone(), 'yyyy-MM-dd'), '', d.name, 'Debt', Number(d.totalLoan || 0), Number(d.paidAmount || 0), 'Remaining: ' + Math.max(0, Number(d.totalLoan || 0) - Number(d.paidAmount || 0))]);
    });

    if (mode === 'replace_all' && data.kind !== 'ping') {
      sheet.clearContents();
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      if (rows.length > 0) {
        sheet.getRange(2, 1, rows.length, rows[0].length).setValues(rows);
      }
    } else {
      if (sheet.getLastRow() === 0) {
        sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      }
      if (rows.length > 0) {
        sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows);
      }
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: 'ok', batchId: batchId, count: rows.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: 'error', message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
